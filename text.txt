import { useEffect } from "react";
import "./DrawingLine.css";

export const DrawLine = () => {
  useEffect(() => {
    const svg = document.getElementById("line") as SVGSVGElement | null;
    const path = document.querySelector(".line-running") as SVGPathElement | null;
    if (!svg || !path) {
      console.warn("SVG oder Pfad nicht gefunden!");
      return;
    }

    const length = path.getTotalLength();
    path.style.strokeDasharray = `${length} ${length}`;
    path.style.strokeDashoffset = `${length}`;

    const startAt = 0.0;
    const finishAt = 1;

    let ticking = false;
    const handleScroll = () => {
      if (ticking) return;
      ticking = true;
      requestAnimationFrame(() => {
        const rect = svg.getBoundingClientRect();
        const vh = window.innerHeight;
        const max = rect.height + vh;
        const raw = (vh - rect.top) / max; // ~0..1 Fortschritt, wenn SVG im Viewport

        // Fortschritt zwischen startAt und finishAt einspannen
        let progress = (raw - startAt) / (finishAt - startAt);
        progress = Math.max(0, Math.min(progress, 1));

        const drawLen = length * progress;
        path.style.strokeDashoffset = String(length - drawLen);

        ticking = false;
      });
    };

    handleScroll(); // Initial rechnen
    window.addEventListener("scroll", handleScroll, { passive: true });

    return () => window.removeEventListener("scroll", handleScroll);
  }, []);
